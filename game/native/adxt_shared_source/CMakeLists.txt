cmake_minimum_required(VERSION 3.20)
project(reo_native_adxt_shared_source LANGUAGES CXX)
if(NOT TARGET reo_native_adxt_refill)
    set(REO_NATIVE_ADXT_REFILL_TESTS OFF)
    add_subdirectory(../adxt_refill adxt_refill)
endif()
add_library(reo_native_adxt_shared_source STATIC shared_source.cpp)
target_compile_features(reo_native_adxt_shared_source PUBLIC cxx_std_20)
target_include_directories(reo_native_adxt_shared_source PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(reo_native_adxt_shared_source PUBLIC reo_native_adxt_refill)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(reo_native_adxt_shared_source PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-misleading-indentation)
endif()
option(REO_NATIVE_ADXT_SHARED_SOURCE_TESTS "Build shared encoded-source checks" ON)
if(REO_NATIVE_ADXT_SHARED_SOURCE_TESTS)
    enable_testing()
    add_executable(reo_native_adxt_shared_source_tests shared_source_tests.cpp)
    target_link_libraries(reo_native_adxt_shared_source_tests PRIVATE reo_native_adxt_shared_source)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(reo_native_adxt_shared_source_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-misleading-indentation)
    endif()
    add_test(NAME adxt_shared_source_owner COMMAND reo_native_adxt_shared_source_tests)
    set_tests_properties(adxt_shared_source_owner PROPERTIES TIMEOUT 30)
endif()

cmake_minimum_required(VERSION 3.20)
project(reo_native_adxt_lsc_input LANGUAGES CXX)
if(NOT TARGET reo_native_adx_sequence)
  add_subdirectory(../adx_sequence adx_sequence)
endif()
if(NOT TARGET reo_native_lsc_queue)
  add_subdirectory(../lsc_queue lsc_queue)
endif()
if(NOT TARGET reo_native_adxt_shared_source)
  add_subdirectory(../adxt_shared_source adxt_shared_source)
endif()
add_library(reo_native_adxt_lsc_input STATIC input.cpp)
target_compile_features(reo_native_adxt_lsc_input PUBLIC cxx_std_20)
target_include_directories(reo_native_adxt_lsc_input PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_adxt_lsc_input PUBLIC reo_native_adx_sequence reo_native_lsc_queue reo_native_adxt_shared_source)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(reo_native_adxt_lsc_input PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

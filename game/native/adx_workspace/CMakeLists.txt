cmake_minimum_required(VERSION 3.20)
project(reo_adx_workspace LANGUAGES CXX)
add_library(reo_adx_workspace STATIC adx_workspace.cpp)
target_compile_features(reo_adx_workspace PUBLIC cxx_std_20)
target_include_directories(reo_adx_workspace PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(reo_adx_workspace PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(adx_workspace_tests adx_workspace_tests.cpp)
target_link_libraries(adx_workspace_tests PRIVATE reo_adx_workspace)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(adx_workspace_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME adx_workspace COMMAND adx_workspace_tests)

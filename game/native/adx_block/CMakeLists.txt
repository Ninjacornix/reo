cmake_minimum_required(VERSION 3.20)
project(reo_native_adx_block LANGUAGES CXX)
add_library(reo_native_adx_block STATIC adx_block.cpp)
target_compile_features(reo_native_adx_block PUBLIC cxx_std_20)
target_include_directories(reo_native_adx_block PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(reo_native_adx_block PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(REO_NATIVE_ADX_BLOCK_TESTS "Build isolated keyed mono ADX block checks" ON)
if(REO_NATIVE_ADX_BLOCK_TESTS)
    enable_testing()
    add_executable(reo_native_adx_block_tests adx_block_tests.cpp)
    target_link_libraries(reo_native_adx_block_tests PRIVATE reo_native_adx_block)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(reo_native_adx_block_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME native_adx_block_contract COMMAND reo_native_adx_block_tests)
    set_tests_properties(native_adx_block_contract PROPERTIES TIMEOUT 30)
endif()

cmake_minimum_required(VERSION 3.20)
project(reo_native_audio_clip_loader LANGUAGES CXX)
if(NOT TARGET reo_native_adx_sequence)
    add_subdirectory(../adx_sequence adx_sequence)
endif()
if(NOT TARGET reo_native_tasks)
    add_subdirectory(../tasks tasks)
endif()
add_library(reo_native_audio_clip_loader STATIC loader.cpp)
target_compile_features(reo_native_audio_clip_loader PUBLIC cxx_std_20)
target_include_directories(reo_native_audio_clip_loader PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_audio_clip_loader PUBLIC reo_native_adx_sequence reo_native_tasks)
add_executable(audio_clip_loader_tests loader_tests.cpp)
target_link_libraries(audio_clip_loader_tests PRIVATE reo_native_audio_clip_loader)
foreach(target reo_native_audio_clip_loader audio_clip_loader_tests)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off -frounding-math)
    endif()
endforeach()
enable_testing()
add_test(NAME audio_clip_loader COMMAND audio_clip_loader_tests)
set_tests_properties(audio_clip_loader PROPERTIES TIMEOUT 30)

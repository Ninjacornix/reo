cmake_minimum_required(VERSION 3.20)
project(reo_native_adx_control LANGUAGES CXX)

add_library(reo_native_adx_control STATIC predictor_control.cpp)
target_include_directories(reo_native_adx_control PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(reo_native_adx_control PUBLIC cxx_std_20)
set_target_properties(reo_native_adx_control PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(reo_native_adx_control PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

option(REO_NATIVE_ADX_CONTROL_TESTS "Build independent ADX control checks" ON)
if(REO_NATIVE_ADX_CONTROL_TESTS)
    enable_testing()
    add_executable(reo_native_adx_control_tests predictor_control_tests.cpp)
    set_target_properties(reo_native_adx_control_tests PROPERTIES OUTPUT_NAME adx_control_tests CXX_EXTENSIONS OFF)
    target_link_libraries(reo_native_adx_control_tests PRIVATE reo_native_adx_control)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(reo_native_adx_control_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
        if(MINGW)
            target_link_options(reo_native_adx_control_tests PRIVATE -static-libgcc -static-libstdc++)
        endif()
    endif()
    add_test(NAME adx_control_unit COMMAND reo_native_adx_control_tests --unit)
    set_tests_properties(adx_control_unit PROPERTIES TIMEOUT 20)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    get_filename_component(REO_ADX_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
    if(Python3_FOUND AND EXISTS "${REO_ADX_REPO_ROOT}/game_data/SLUS_207.65")
        add_test(NAME adx_control_original_words COMMAND "${Python3_EXECUTABLE}"
            "${REO_ADX_REPO_ROOT}/tools/native_reconstruction/compare_adx_control.py"
            --executable "$<TARGET_FILE:reo_native_adx_control_tests>"
            --cxx "${CMAKE_CXX_COMPILER}"
            --output "${REO_ADX_REPO_ROOT}/build-native-sjuni/adx-control-cmake-${CMAKE_CXX_COMPILER_ID}/adx-control-checks")
        set_tests_properties(adx_control_original_words PROPERTIES TIMEOUT 60)
    endif()
endif()

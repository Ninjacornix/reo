cmake_minimum_required(VERSION 3.20)
project(reo_native_adxt_clocked_player LANGUAGES CXX)
if(NOT TARGET reo_native_stream_window_v2_playback_r2)
  add_subdirectory(../adx_window/v2/playback_r2 stream_window_v2_playback_r2)
endif()
if(NOT TARGET reo_native_adxt_output_session)
  add_subdirectory(../adxt_output_session adxt_output_session)
endif()
if(NOT TARGET reo_native_adxt_estimated_clock)
  add_subdirectory(../adxt_estimated_clock adxt_estimated_clock)
endif()
add_library(reo_native_adxt_clocked_player STATIC player.cpp)
target_compile_features(reo_native_adxt_clocked_player PUBLIC cxx_std_20)
target_include_directories(reo_native_adxt_clocked_player PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_adxt_clocked_player PUBLIC reo_native_stream_window_v2_playback_r2 reo_native_adxt_output_session reo_native_adxt_estimated_clock)
add_executable(adxt_clocked_player_tests player_tests.cpp)
target_link_libraries(adxt_clocked_player_tests PRIVATE reo_native_adxt_clocked_player)
add_executable(adxt_clocked_player_inherited_tests inherited_tests.cpp)
target_link_libraries(adxt_clocked_player_inherited_tests PRIVATE reo_native_adxt_clocked_player)
foreach(target reo_native_adxt_clocked_player adxt_clocked_player_tests adxt_clocked_player_inherited_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off -frounding-math)
  endif()
endforeach()
enable_testing()
add_test(NAME adxt_clocked_player COMMAND adxt_clocked_player_tests)
add_test(NAME adxt_clocked_player_inherited COMMAND adxt_clocked_player_inherited_tests)
set_tests_properties(adxt_clocked_player PROPERTIES TIMEOUT 60)

cmake_minimum_required(VERSION 3.20)
project(reo_native_movie_position LANGUAGES CXX)
add_library(reo_native_movie_position STATIC movie_position.cpp)
target_include_directories(reo_native_movie_position PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(reo_native_movie_position PUBLIC cxx_std_20)
option(REO_NATIVE_MOVIE_POSITION_TESTS "Build isolated picture-position checks" ON)
if(REO_NATIVE_MOVIE_POSITION_TESTS)
    enable_testing()
    add_executable(reo_native_movie_position_tests movie_position_tests.cpp)
    target_link_libraries(reo_native_movie_position_tests PRIVATE reo_native_movie_position)
    add_test(NAME movie_position_unit COMMAND reo_native_movie_position_tests)
endif()

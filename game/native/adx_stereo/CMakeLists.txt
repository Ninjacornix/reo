cmake_minimum_required(VERSION 3.20)
project(reo_native_adx_stereo LANGUAGES CXX)
add_library(reo_native_adx_stereo STATIC adx_stereo.cpp)
target_compile_features(reo_native_adx_stereo PUBLIC cxx_std_20)
target_include_directories(reo_native_adx_stereo PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(reo_native_adx_stereo PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(REO_NATIVE_ADX_STEREO_TESTS "Build isolated keyed stereo checks" ON)
if(REO_NATIVE_ADX_STEREO_TESTS)
    enable_testing()
    add_executable(reo_native_adx_stereo_tests adx_stereo_tests.cpp)
    target_link_libraries(reo_native_adx_stereo_tests PRIVATE reo_native_adx_stereo)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(reo_native_adx_stereo_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME native_adx_stereo_contract COMMAND reo_native_adx_stereo_tests)
    set_tests_properties(native_adx_stereo_contract PROPERTIES TIMEOUT 30)
endif()

cmake_minimum_required(VERSION 3.20)
project(reo_adx_pcm_store LANGUAGES CXX)
if(NOT TARGET reo_native_sjrb)
  add_subdirectory(../sjrb sjrb)
endif()
add_library(reo_adx_pcm_store STATIC pcm_store.cpp)
target_compile_features(reo_adx_pcm_store PUBLIC cxx_std_20)
target_link_libraries(reo_adx_pcm_store PUBLIC reo_native_sjrb)
target_include_directories(reo_adx_pcm_store PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(reo_adx_pcm_store PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(adx_pcm_store_tests pcm_store_tests.cpp)
target_link_libraries(adx_pcm_store_tests PRIVATE reo_adx_pcm_store)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(adx_pcm_store_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME adx_pcm_store COMMAND adx_pcm_store_tests)

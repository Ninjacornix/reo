cmake_minimum_required(VERSION 3.20)
project(reo_native_adxt_control_session LANGUAGES CXX)
if(NOT TARGET reo_native_adxt_phased_player)
  add_subdirectory(../adxt_phased_player adxt_phased_player)
endif()
add_library(reo_native_adxt_control_session STATIC control_session.cpp)
target_compile_features(reo_native_adxt_control_session PUBLIC cxx_std_20)
target_include_directories(reo_native_adxt_control_session PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_adxt_control_session PUBLIC reo_native_adxt_phased_player)
add_executable(adxt_control_session_tests control_session_tests.cpp)
target_link_libraries(adxt_control_session_tests PRIVATE reo_native_adxt_control_session)
foreach(target reo_native_adxt_control_session adxt_control_session_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off -frounding-math)
  endif()
endforeach()
enable_testing()
add_test(NAME adxt_control_session COMMAND adxt_control_session_tests)
set_tests_properties(adxt_control_session PROPERTIES TIMEOUT 60)
add_executable(adxt_lsc_input_owner_tests ../adxt_lsc_input/owner_tests.cpp)
target_link_libraries(adxt_lsc_input_owner_tests PRIVATE reo_native_adxt_control_session)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(adxt_lsc_input_owner_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off -frounding-math)
endif()
add_test(NAME adxt_lsc_input_owner COMMAND adxt_lsc_input_owner_tests)
set_tests_properties(adxt_lsc_input_owner PROPERTIES TIMEOUT 60)
add_executable(audio_mix_tail_owner_tests ../audio_mix_tail/owner_tests.cpp)
target_link_libraries(audio_mix_tail_owner_tests PRIVATE reo_native_adxt_control_session)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(audio_mix_tail_owner_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off -frounding-math)
endif()
add_test(NAME audio_mix_tail_owner COMMAND audio_mix_tail_owner_tests)
set_tests_properties(audio_mix_tail_owner PROPERTIES TIMEOUT 60)

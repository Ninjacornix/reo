cmake_minimum_required(VERSION 3.20)
project(reo_native_afs_source_range LANGUAGES CXX)
if(NOT TARGET reo_native_resources)
  add_subdirectory(../resources resources)
endif()
add_library(reo_native_afs_source_range STATIC afs_source_range.cpp)
target_compile_features(reo_native_afs_source_range PUBLIC cxx_std_20)
target_include_directories(reo_native_afs_source_range PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_afs_source_range PUBLIC reo_native_resources)
add_executable(afs_source_range_tests afs_source_range_tests.cpp)
target_link_libraries(afs_source_range_tests PRIVATE reo_native_afs_source_range)
foreach(target reo_native_afs_source_range afs_source_range_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
enable_testing()
add_test(NAME afs_source_range COMMAND afs_source_range_tests)

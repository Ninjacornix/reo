cmake_minimum_required(VERSION 3.20)
project(reo_native_adxt_estimated_clock LANGUAGES CXX)
if(NOT TARGET reo_adxt_integer_clock)
  add_subdirectory(../adxt_integer_clock adxt_integer_clock)
endif()
add_library(reo_native_adxt_estimated_clock STATIC estimated_clock.cpp)
target_compile_features(reo_native_adxt_estimated_clock PUBLIC cxx_std_20)
target_include_directories(reo_native_adxt_estimated_clock PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_adxt_estimated_clock PUBLIC reo_adxt_integer_clock)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(reo_native_adxt_estimated_clock PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off -frounding-math)
endif()
add_executable(estimated_clock_tests estimated_clock_tests.cpp)
target_link_libraries(estimated_clock_tests PRIVATE reo_native_adxt_estimated_clock)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(estimated_clock_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off -frounding-math)
endif()
enable_testing()
add_test(NAME adxt_estimated_clock COMMAND estimated_clock_tests)

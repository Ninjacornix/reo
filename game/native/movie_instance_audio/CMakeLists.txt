cmake_minimum_required(VERSION 3.20)
project(reo_native_movie_instance_audio LANGUAGES CXX)
if(NOT TARGET reo_native_movie_audio_clock)
  add_subdirectory(../movie_audio_clock movie_audio_clock)
endif()
if(NOT TARGET reo_native_adxt_control_owner)
  add_subdirectory(../adxt_control_owner adxt_control_owner)
endif()
add_library(reo_native_movie_instance_audio STATIC source.cpp)
target_compile_features(reo_native_movie_instance_audio PUBLIC cxx_std_20)
target_include_directories(reo_native_movie_instance_audio PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_movie_instance_audio PUBLIC reo_native_movie_audio_clock reo_native_adxt_control_owner)
add_executable(movie_instance_audio_tests source_tests.cpp)
target_link_libraries(movie_instance_audio_tests PRIVATE reo_native_movie_instance_audio)
foreach(target reo_native_movie_instance_audio movie_instance_audio_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off -frounding-math)
  endif()
endforeach()
enable_testing()
add_test(NAME movie_instance_audio COMMAND movie_instance_audio_tests)
set_tests_properties(movie_instance_audio PROPERTIES TIMEOUT 60)

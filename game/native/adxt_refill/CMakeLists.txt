cmake_minimum_required(VERSION 3.20)
project(reo_native_adxt_refill LANGUAGES CXX)
if(NOT TARGET reo_native_sjrb)
    set(REO_NATIVE_SJRB_TESTS OFF)
    add_subdirectory(../sjrb sjrb)
endif()
if(NOT TARGET reo_native_resource_io)
    set(REO_NATIVE_RESOURCE_IO_TESTS OFF)
    add_subdirectory(../resource_io resource_io)
endif()
add_library(reo_native_adxt_refill STATIC adxt_refill.cpp)
target_compile_features(reo_native_adxt_refill PUBLIC cxx_std_20)
target_include_directories(reo_native_adxt_refill PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(reo_native_adxt_refill PUBLIC reo_native_sjrb reo_native_resource_io)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(reo_native_adxt_refill PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(REO_NATIVE_ADXT_REFILL_TESTS "Build encoded-source ownership checks" ON)
if(REO_NATIVE_ADXT_REFILL_TESTS)
    enable_testing()
    add_executable(reo_native_adxt_refill_tests adxt_refill_tests.cpp)
    target_link_libraries(reo_native_adxt_refill_tests PRIVATE reo_native_adxt_refill)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(reo_native_adxt_refill_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME adxt_refill_owner COMMAND reo_native_adxt_refill_tests)
    set_tests_properties(adxt_refill_owner PROPERTIES TIMEOUT 30)
endif()

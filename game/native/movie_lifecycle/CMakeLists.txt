cmake_minimum_required(VERSION 3.20)
project(reo_native_movie_lifecycle LANGUAGES CXX)
foreach(component movie_timing_context resource_io)
  if(NOT TARGET reo_native_${component})
    add_subdirectory(../${component} ${component})
  endif()
endforeach()
add_library(reo_native_movie_lifecycle STATIC policy.cpp session.cpp prepared_graph.cpp)
target_compile_features(reo_native_movie_lifecycle PUBLIC cxx_std_20)
target_include_directories(reo_native_movie_lifecycle PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_movie_lifecycle PUBLIC reo_native_movie_timing_context reo_native_resource_io)
add_executable(movie_lifecycle_tests session_tests.cpp)
target_link_libraries(movie_lifecycle_tests PRIVATE reo_native_movie_lifecycle)
foreach(target reo_native_movie_lifecycle movie_lifecycle_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off -frounding-math)
  endif()
endforeach()
enable_testing()
add_test(NAME movie_lifecycle COMMAND movie_lifecycle_tests)
set_tests_properties(movie_lifecycle PROPERTIES TIMEOUT 60)

cmake_minimum_required(VERSION 3.20)
project(reo_native_movie_clock_initialize LANGUAGES CXX)
add_library(reo_native_movie_clock_initialize STATIC initialize.cpp)
target_compile_features(reo_native_movie_clock_initialize PUBLIC cxx_std_20)
target_include_directories(reo_native_movie_clock_initialize PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(movie_clock_initialize_tests initialize_tests.cpp)
target_link_libraries(movie_clock_initialize_tests PRIVATE reo_native_movie_clock_initialize)
foreach(target reo_native_movie_clock_initialize movie_clock_initialize_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
enable_testing()
add_test(NAME movie_clock_initialize COMMAND movie_clock_initialize_tests)
set_tests_properties(movie_clock_initialize PROPERTIES TIMEOUT 30)

cmake_minimum_required(VERSION 3.20)
project(reo_native_movie_assets LANGUAGES CXX)
if(NOT TARGET reo_native_resources_windows)
  add_subdirectory(../resources_windows native_movie_resources)
endif()
add_library(reo_native_movie_assets STATIC catalogue.cpp)
target_compile_features(reo_native_movie_assets PUBLIC cxx_std_20)
target_include_directories(reo_native_movie_assets PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_movie_assets PUBLIC reo_native_resources_windows runtimeobject)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(reo_native_movie_assets PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

cmake_minimum_required(VERSION 3.20)
project(reo_native_pcm_bridge LANGUAGES CXX)

if(NOT TARGET reo_native_pcm)
    add_subdirectory(../pcm pcm_core)
endif()
find_package(Threads REQUIRED)
add_library(reo_native_pcm_bridge STATIC session_bridge.cpp)
target_include_directories(reo_native_pcm_bridge PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(reo_native_pcm_bridge PUBLIC cxx_std_20)
target_link_libraries(reo_native_pcm_bridge PUBLIC reo_native_pcm Threads::Threads)
if(NOT MSVC)
    target_compile_options(reo_native_pcm_bridge PRIVATE -ffp-contract=off)
endif()

option(REO_NATIVE_PCM_BRIDGE_TESTS "Build isolated raw-ID PCM session checks" ON)
if(REO_NATIVE_PCM_BRIDGE_TESTS)
    enable_testing()
    add_executable(reo_native_pcm_bridge_tests session_bridge_tests.cpp)
    target_link_libraries(reo_native_pcm_bridge_tests PRIVATE reo_native_pcm_bridge)
    add_test(NAME pcm_session_bridge_contract COMMAND reo_native_pcm_bridge_tests)
    set_tests_properties(pcm_session_bridge_contract PROPERTIES TIMEOUT 30)
endif()

cmake_minimum_required(VERSION 3.20)
project(reo_native_adxt_player LANGUAGES CXX)
if(NOT TARGET reo_native_stream_window_v2_playback_r2)
  add_subdirectory(../adx_window/v2/playback_r2 stream_window_v2_playback_r2)
endif()
if(NOT TARGET reo_native_pcm)
  add_subdirectory(../pcm pcm)
endif()
if(NOT TARGET reo_native_audio_output)
  add_subdirectory(../audio_output audio_output)
endif()
add_library(reo_native_adxt_player STATIC player.cpp)
target_compile_features(reo_native_adxt_player PUBLIC cxx_std_20)
target_include_directories(reo_native_adxt_player PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_adxt_player PUBLIC reo_native_stream_window_v2_playback_r2
  reo_native_pcm reo_native_audio_output)
add_executable(adxt_player_tests player_tests.cpp)
target_link_libraries(adxt_player_tests PRIVATE reo_native_adxt_player)
foreach(target reo_native_adxt_player adxt_player_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off -frounding-math)
  endif()
endforeach()
enable_testing()
add_test(NAME adxt_player COMMAND adxt_player_tests)
set_tests_properties(adxt_player PROPERTIES TIMEOUT 60)

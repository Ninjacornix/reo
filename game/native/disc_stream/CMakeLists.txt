cmake_minimum_required(VERSION 3.20)
project(reo_native_disc_stream LANGUAGES CXX)
include(CTest)

add_library(reo_native_disc_stream STATIC disc_stream.cpp)
target_compile_features(reo_native_disc_stream PUBLIC cxx_std_20)
target_include_directories(reo_native_disc_stream PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
    target_compile_options(reo_native_disc_stream PRIVATE /W4 /WX)
else()
    target_compile_options(reo_native_disc_stream PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
    add_executable(reo_native_disc_stream_tests disc_stream_tests.cpp)
    target_link_libraries(reo_native_disc_stream_tests PRIVATE reo_native_disc_stream)
    if(MSVC)
        target_compile_options(reo_native_disc_stream_tests PRIVATE /W4 /WX)
    else()
        target_compile_options(reo_native_disc_stream_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME native_disc_stream COMMAND reo_native_disc_stream_tests)
endif()

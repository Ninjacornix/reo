cmake_minimum_required(VERSION 3.20)
project(reo_native_tasks LANGUAGES CXX)

find_package(Threads REQUIRED)
add_library(reo_native_tasks STATIC serial_worker.cpp)
target_include_directories(reo_native_tasks PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(reo_native_tasks PUBLIC cxx_std_20)
target_link_libraries(reo_native_tasks PUBLIC Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(reo_native_tasks PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

option(REO_NATIVE_TASKS_TESTS "Build standalone serial worker lifecycle checks" ON)
if(REO_NATIVE_TASKS_TESTS)
    enable_testing()
    add_executable(reo_native_tasks_tests serial_worker_tests.cpp)
    target_link_libraries(reo_native_tasks_tests PRIVATE reo_native_tasks)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(reo_native_tasks_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME serial_worker_contract COMMAND reo_native_tasks_tests)
    set_tests_properties(serial_worker_contract PROPERTIES TIMEOUT 30)
endif()

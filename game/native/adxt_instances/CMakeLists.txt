cmake_minimum_required(VERSION 3.20)
project(reo_native_adxt_instances LANGUAGES CXX)
if(NOT TARGET reo_native_adxt_control_session)
  add_subdirectory(../adxt_control_session adxt_control_session)
endif()
add_library(reo_native_adxt_instances STATIC instances.cpp)
target_compile_features(reo_native_adxt_instances PUBLIC cxx_std_20)
target_include_directories(reo_native_adxt_instances PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_adxt_instances PUBLIC reo_native_adxt_control_session)
add_executable(adxt_instances_tests instances_tests.cpp)
target_link_libraries(adxt_instances_tests PRIVATE reo_native_adxt_instances)
foreach(target reo_native_adxt_instances adxt_instances_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off -frounding-math)
  endif()
endforeach()
enable_testing()
add_test(NAME adxt_instances COMMAND adxt_instances_tests)
set_tests_properties(adxt_instances PROPERTIES TIMEOUT 60)
add_executable(adxt_instance_game_control_tests game_control_tests.cpp)
target_link_libraries(adxt_instance_game_control_tests PRIVATE reo_native_adxt_instances)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(adxt_instance_game_control_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off -frounding-math)
endif()
add_test(NAME adxt_instance_game_controls COMMAND adxt_instance_game_control_tests)
set_tests_properties(adxt_instance_game_controls PROPERTIES TIMEOUT 60)

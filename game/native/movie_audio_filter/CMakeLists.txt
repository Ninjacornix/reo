cmake_minimum_required(VERSION 3.20)
project(reo_native_movie_audio_filter LANGUAGES CXX)
add_library(reo_native_movie_audio_filter STATIC filter.cpp)
target_compile_features(reo_native_movie_audio_filter PUBLIC cxx_std_20)
target_include_directories(reo_native_movie_audio_filter PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(movie_audio_filter_tests filter_tests.cpp)
target_link_libraries(movie_audio_filter_tests PRIVATE reo_native_movie_audio_filter)
foreach(target reo_native_movie_audio_filter movie_audio_filter_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
enable_testing()
add_test(NAME movie_audio_filter COMMAND movie_audio_filter_tests)
set_tests_properties(movie_audio_filter PROPERTIES TIMEOUT 30)

cmake_minimum_required(VERSION 3.20)
project(reo_native_resource_io LANGUAGES CXX)
find_package(Threads REQUIRED)

# Usable standalone or added after its two independent native dependencies.
if(NOT TARGET reo_native_resources)
    set(REO_NATIVE_RESOURCES_TESTS OFF)
    add_subdirectory(../resources resources)
endif()
if(NOT TARGET reo_native_tasks)
    set(REO_NATIVE_TASKS_TESTS OFF)
    add_subdirectory(../tasks tasks)
endif()
add_library(reo_native_resource_io STATIC async_read_owner.cpp)
target_include_directories(reo_native_resource_io PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(reo_native_resource_io PUBLIC cxx_std_20)
target_link_libraries(reo_native_resource_io PUBLIC reo_native_resources reo_native_tasks Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(reo_native_resource_io PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(REO_NATIVE_RESOURCE_IO_TESTS "Build owned asynchronous resource-read checks" ON)
if(REO_NATIVE_RESOURCE_IO_TESTS)
    enable_testing()
    add_executable(reo_native_resource_io_tests async_read_owner_tests.cpp)
    target_link_libraries(reo_native_resource_io_tests PRIVATE reo_native_resource_io)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(reo_native_resource_io_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME native_resource_io_contract COMMAND reo_native_resource_io_tests)
    set_tests_properties(native_resource_io_contract PROPERTIES TIMEOUT 30)
endif()

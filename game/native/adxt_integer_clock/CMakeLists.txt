cmake_minimum_required(VERSION 3.20)
project(reo_adxt_integer_clock LANGUAGES CXX)
add_library(reo_adxt_integer_clock STATIC integer_clock.cpp)
target_compile_features(reo_adxt_integer_clock PUBLIC cxx_std_20)
target_include_directories(reo_adxt_integer_clock PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(adxt_integer_clock_tests integer_clock_tests.cpp)
target_link_libraries(adxt_integer_clock_tests PRIVATE reo_adxt_integer_clock)
foreach(target reo_adxt_integer_clock adxt_integer_clock_tests)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
enable_testing()
add_test(NAME adxt_integer_clock COMMAND adxt_integer_clock_tests)

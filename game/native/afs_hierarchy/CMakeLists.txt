cmake_minimum_required(VERSION 3.20)
project(reo_native_afs_hierarchy LANGUAGES CXX)
if(NOT TARGET reo_native_afs_source_range)
  add_subdirectory(../afs_source_range afs_source_range)
endif()
add_library(reo_native_afs_hierarchy STATIC hierarchy.cpp)
target_compile_features(reo_native_afs_hierarchy PUBLIC cxx_std_20)
target_include_directories(reo_native_afs_hierarchy PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_afs_hierarchy PUBLIC reo_native_afs_source_range)
add_executable(afs_hierarchy_tests hierarchy_tests.cpp)
target_link_libraries(afs_hierarchy_tests PRIVATE reo_native_afs_hierarchy)
foreach(target reo_native_afs_hierarchy afs_hierarchy_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
enable_testing()
add_test(NAME afs_hierarchy COMMAND afs_hierarchy_tests)

cmake_minimum_required(VERSION 3.20)
project(reo_native_lsc_queue LANGUAGES CXX)
add_library(reo_native_lsc_queue STATIC queue.cpp)
target_compile_features(reo_native_lsc_queue PUBLIC cxx_std_20)
target_include_directories(reo_native_lsc_queue PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(lsc_queue_tests queue_tests.cpp)
target_link_libraries(lsc_queue_tests PRIVATE reo_native_lsc_queue)
foreach(target reo_native_lsc_queue lsc_queue_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
enable_testing()
add_test(NAME lsc_queue COMMAND lsc_queue_tests)
set_tests_properties(lsc_queue PROPERTIES TIMEOUT 30)

cmake_minimum_required(VERSION 3.20)
project(reo_native_adxt_output_session LANGUAGES CXX)
if(NOT TARGET reo_native_pcm)
  add_subdirectory(../pcm pcm)
endif()
if(NOT TARGET reo_native_audio_output)
  add_subdirectory(../audio_output audio_output)
endif()
add_library(reo_native_adxt_output_session STATIC session.cpp)
target_compile_features(reo_native_adxt_output_session PUBLIC cxx_std_20)
target_include_directories(reo_native_adxt_output_session PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_adxt_output_session PUBLIC reo_native_pcm reo_native_audio_output)
add_executable(adxt_output_session_tests session_tests.cpp)
target_link_libraries(adxt_output_session_tests PRIVATE reo_native_adxt_output_session)
foreach(target reo_native_adxt_output_session adxt_output_session_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off -fno-fast-math)
  endif()
endforeach()
enable_testing()
add_test(NAME adxt_output_session COMMAND adxt_output_session_tests)
set_tests_properties(adxt_output_session PROPERTIES TIMEOUT 30)

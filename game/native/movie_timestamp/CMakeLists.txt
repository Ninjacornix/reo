cmake_minimum_required(VERSION 3.20)
project(reo_native_movie_timestamp LANGUAGES CXX)
add_library(reo_native_movie_timestamp STATIC timestamp.cpp timecode.cpp)
target_compile_features(reo_native_movie_timestamp PUBLIC cxx_std_20)
target_include_directories(reo_native_movie_timestamp PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(movie_timestamp_tests timestamp_tests.cpp)
target_link_libraries(movie_timestamp_tests PRIVATE reo_native_movie_timestamp)
foreach(target reo_native_movie_timestamp movie_timestamp_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
  endif()
endforeach()
enable_testing()
add_test(NAME movie_timestamp COMMAND movie_timestamp_tests)

cmake_minimum_required(VERSION 3.20)
project(reo_native_ram_capture LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(reo_native_ram_capture STATIC cpu_authority_gate.cpp)
target_compile_features(reo_native_ram_capture PUBLIC cxx_std_20)
target_include_directories(reo_native_ram_capture PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(reo_native_ram_capture PUBLIC Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(reo_native_ram_capture PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(REO_NATIVE_RAM_CAPTURE_TESTS "Build CPU capture/graphics admission checks" ON)
if(REO_NATIVE_RAM_CAPTURE_TESTS)
    enable_testing()
    add_executable(reo_native_ram_capture_tests cpu_authority_gate_tests.cpp)
    target_link_libraries(reo_native_ram_capture_tests PRIVATE reo_native_ram_capture)
    add_test(NAME ram_capture_gate COMMAND reo_native_ram_capture_tests)
endif()

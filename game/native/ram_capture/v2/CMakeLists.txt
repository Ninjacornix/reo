cmake_minimum_required(VERSION 3.20)
project(reo_native_ram_capture_v2 LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(reo_native_ram_capture_v2 STATIC cpu_authority_gate.cpp)
target_compile_features(reo_native_ram_capture_v2 PUBLIC cxx_std_20)
target_include_directories(reo_native_ram_capture_v2 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(reo_native_ram_capture_v2 PUBLIC Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(reo_native_ram_capture_v2 PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-misleading-indentation)
endif()
option(REO_NATIVE_RAM_CAPTURE_V2_TESTS "Build cooperating reader/writer admission checks" ON)
if(REO_NATIVE_RAM_CAPTURE_V2_TESTS)
    enable_testing()
    add_executable(reo_native_ram_capture_v2_tests cpu_authority_gate_tests.cpp)
    target_link_libraries(reo_native_ram_capture_v2_tests PRIVATE reo_native_ram_capture_v2)
    add_test(NAME ram_capture_writer_gate COMMAND reo_native_ram_capture_v2_tests)
    set_tests_properties(ram_capture_writer_gate PROPERTIES TIMEOUT 30)
endif()

cmake_minimum_required(VERSION 3.20)
project(reo_native_adx_sequence LANGUAGES CXX)
if(NOT TARGET reo_native_adx_header)
  add_subdirectory(../adx_header adx_header)
endif()
add_library(reo_native_adx_sequence STATIC sequence.cpp)
target_compile_features(reo_native_adx_sequence PUBLIC cxx_std_20)
target_include_directories(reo_native_adx_sequence PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_adx_sequence PUBLIC reo_native_adx_header)
add_executable(adx_sequence_tests sequence_tests.cpp)
target_link_libraries(adx_sequence_tests PRIVATE reo_native_adx_sequence)
foreach(target reo_native_adx_sequence adx_sequence_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off -frounding-math)
  endif()
endforeach()
enable_testing()
add_test(NAME adx_sequence COMMAND adx_sequence_tests)

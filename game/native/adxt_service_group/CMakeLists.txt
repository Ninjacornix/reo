cmake_minimum_required(VERSION 3.20)
project(reo_native_adxt_service_group LANGUAGES CXX)
add_library(reo_native_adxt_service_group STATIC service_group.cpp)
target_compile_features(reo_native_adxt_service_group PUBLIC cxx_std_20)
target_include_directories(reo_native_adxt_service_group PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(adxt_service_group_tests service_group_tests.cpp)
target_link_libraries(adxt_service_group_tests PRIVATE reo_native_adxt_service_group)
foreach(target reo_native_adxt_service_group adxt_service_group_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
enable_testing()
add_test(NAME adxt_service_group COMMAND adxt_service_group_tests)

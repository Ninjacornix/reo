cmake_minimum_required(VERSION 3.20)
project(reo_native_gameplay_tasks LANGUAGES CXX)
enable_testing()
add_library(reo_native_gameplay_tasks STATIC tasks.cpp events.cpp)
target_compile_features(reo_native_gameplay_tasks PUBLIC cxx_std_20)
target_include_directories(reo_native_gameplay_tasks PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(gameplay_tasks_tests tests.cpp)
target_link_libraries(gameplay_tasks_tests PRIVATE reo_native_gameplay_tasks)
add_test(NAME native_gameplay_tasks COMMAND gameplay_tasks_tests)

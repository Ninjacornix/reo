cmake_minimum_required(VERSION 3.20)
project(reo_native_movie_selected_clock LANGUAGES CXX)
if(NOT TARGET reo_native_movie_counter_clock)
  add_subdirectory(../movie_counter_clock movie_counter_clock)
endif()
add_library(reo_native_movie_selected_clock STATIC clock.cpp)
target_compile_features(reo_native_movie_selected_clock PUBLIC cxx_std_20)
target_include_directories(reo_native_movie_selected_clock PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_movie_selected_clock PUBLIC reo_native_movie_counter_clock)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(reo_native_movie_selected_clock PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(movie_selected_clock_tests clock_tests.cpp)
target_link_libraries(movie_selected_clock_tests PRIVATE reo_native_movie_selected_clock)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(movie_selected_clock_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME movie_selected_clock COMMAND movie_selected_clock_tests)
set_tests_properties(movie_selected_clock PROPERTIES TIMEOUT 30)

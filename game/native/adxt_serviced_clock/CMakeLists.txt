cmake_minimum_required(VERSION 3.20)
project(reo_native_adxt_serviced_clock LANGUAGES CXX)
if(NOT TARGET reo_native_adxt_estimated_clock)
  add_subdirectory(../adxt_estimated_clock adxt_estimated_clock)
endif()
add_library(reo_native_adxt_serviced_clock STATIC completion.cpp)
target_compile_features(reo_native_adxt_serviced_clock PUBLIC cxx_std_20)
target_include_directories(reo_native_adxt_serviced_clock PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_adxt_serviced_clock PUBLIC reo_native_adxt_estimated_clock)
add_executable(adxt_serviced_clock_tests completion_tests.cpp)
target_link_libraries(adxt_serviced_clock_tests PRIVATE reo_native_adxt_serviced_clock)
foreach(target reo_native_adxt_serviced_clock adxt_serviced_clock_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off -frounding-math)
  endif()
endforeach()
enable_testing()
add_test(NAME adxt_serviced_clock COMMAND adxt_serviced_clock_tests)

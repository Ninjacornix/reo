cmake_minimum_required(VERSION 3.20)
project(reo_native_audio_banks LANGUAGES CXX)

find_package(Threads REQUIRED)
add_library(reo_native_audio_banks STATIC algorithms.cpp session.cpp)
target_include_directories(reo_native_audio_banks PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(reo_native_audio_banks PUBLIC cxx_std_20)
target_link_libraries(reo_native_audio_banks PUBLIC Threads::Threads)
if(MSVC)
    target_compile_options(reo_native_audio_banks PRIVATE /fp:strict)
else()
    target_compile_options(reo_native_audio_banks PRIVATE -fno-fast-math -ffp-contract=off)
endif()

option(REO_NATIVE_AUDIO_BANK_TESTS "Build isolated bank session contract checks" ON)
if(REO_NATIVE_AUDIO_BANK_TESTS)
    enable_testing()
    add_executable(reo_native_audio_bank_tests session_tests.cpp)
    target_link_libraries(reo_native_audio_bank_tests PRIVATE reo_native_audio_banks)
    add_test(NAME audio_bank_session_contract COMMAND reo_native_audio_bank_tests)
    set_tests_properties(audio_bank_session_contract PROPERTIES TIMEOUT 30)
endif()

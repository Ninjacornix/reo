cmake_minimum_required(VERSION 3.20)
project(reo_native_audio_banks_v2 LANGUAGES CXX)

find_package(Threads REQUIRED)
add_library(reo_native_audio_banks_v2 STATIC algorithms.cpp session.cpp)
target_include_directories(reo_native_audio_banks_v2 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(reo_native_audio_banks_v2 PUBLIC cxx_std_20)
target_link_libraries(reo_native_audio_banks_v2 PUBLIC Threads::Threads)
if(MSVC)
    target_compile_options(reo_native_audio_banks_v2 PRIVATE /fp:strict)
else()
    target_compile_options(reo_native_audio_banks_v2 PRIVATE -fno-fast-math -ffp-contract=off)
endif()

option(REO_NATIVE_AUDIO_BANK_V2_TESTS "Build isolated budgeted bank session checks" ON)
if(REO_NATIVE_AUDIO_BANK_V2_TESTS)
    enable_testing()
    add_executable(reo_native_audio_bank_v2_tests budget_tests.cpp)
    target_link_libraries(reo_native_audio_bank_v2_tests PRIVATE reo_native_audio_banks_v2)
    add_test(NAME audio_bank_budget_contract COMMAND reo_native_audio_bank_v2_tests)
    set_tests_properties(audio_bank_budget_contract PROPERTIES TIMEOUT 30)
endif()

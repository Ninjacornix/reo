cmake_minimum_required(VERSION 3.20)
project(reo_native_movie_service LANGUAGES CXX)
add_library(reo_native_movie_service STATIC service.cpp)
target_compile_features(reo_native_movie_service PUBLIC cxx_std_20)
target_include_directories(reo_native_movie_service PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(movie_service_tests service_tests.cpp)
target_link_libraries(movie_service_tests PRIVATE reo_native_movie_service)
foreach(target reo_native_movie_service movie_service_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
enable_testing()
add_test(NAME movie_service COMMAND movie_service_tests)

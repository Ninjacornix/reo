cmake_minimum_required(VERSION 3.20)
project(reo_native_adx_window LANGUAGES CXX)
foreach(dep IN ITEMS sjrb adx_block adx_stereo)
    if(NOT TARGET reo_native_${dep})
        add_library(reo_native_${dep} STATIC ../${dep}/${dep}.cpp)
        target_compile_features(reo_native_${dep} PUBLIC cxx_std_20)
    endif()
endforeach()
if(NOT TARGET reo_native_adx_control)
    add_library(reo_native_adx_control STATIC ../adx_control/predictor_control.cpp ../adx_control/predictor_service.cpp)
    target_compile_features(reo_native_adx_control PUBLIC cxx_std_20)
endif()
add_library(reo_native_adx_window STATIC adx_window.cpp)
target_include_directories(reo_native_adx_window PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(reo_native_adx_window PUBLIC cxx_std_20)
target_link_libraries(reo_native_adx_window PUBLIC reo_native_sjrb reo_native_adx_block reo_native_adx_stereo reo_native_adx_control)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(reo_native_adx_window PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(REO_NATIVE_ADX_WINDOW_TESTS "Build standalone typed ADX window checks" ON)
if(REO_NATIVE_ADX_WINDOW_TESTS)
    enable_testing()
    add_executable(reo_native_adx_window_tests adx_window_tests.cpp)
    target_link_libraries(reo_native_adx_window_tests PRIVATE reo_native_adx_window)
    add_test(NAME adx_window_unit COMMAND reo_native_adx_window_tests)
endif()

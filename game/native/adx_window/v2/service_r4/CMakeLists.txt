cmake_minimum_required(VERSION 3.20)
project(reo_native_stream_window_service_r4 LANGUAGES CXX)
foreach(dep IN ITEMS adxt_shared_source adx_header adx_workspace adxt_playback)
  if(NOT TARGET reo_native_${dep} AND NOT TARGET reo_${dep})
    add_subdirectory(../../../${dep} ${dep})
  endif()
endforeach()
add_library(reo_native_stream_window_service_r4 STATIC stream_window.cpp)
target_compile_features(reo_native_stream_window_service_r4 PUBLIC cxx_std_20)
target_include_directories(reo_native_stream_window_service_r4 PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_stream_window_service_r4 PUBLIC reo_native_adxt_shared_source
  reo_native_adx_header reo_adx_workspace reo_native_adxt_playback)

# Re-run the frozen inherited cases against this distinct class/ABI. The
# inherited file and all historical headers/implementations remain unchanged.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../playback_r2/inherited_tests.cpp" inherited)
string(REPLACE "#include \"stream_window.hpp\"" "#include \"${CMAKE_CURRENT_SOURCE_DIR}/stream_window.hpp\"" inherited "${inherited}")
string(REPLACE "reo::native::adx_window::v2" "reo::native::adx_window::service_r4" inherited "${inherited}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/inherited_tests.cpp" "${inherited}")
add_executable(stream_window_service_r4_inherited_tests "${CMAKE_CURRENT_BINARY_DIR}/inherited_tests.cpp")
add_executable(stream_window_service_r4_quantum_tests quantum_tests.cpp)
foreach(target stream_window_service_r4_inherited_tests stream_window_service_r4_quantum_tests)
  target_link_libraries(${target} PRIVATE reo_native_stream_window_service_r4)
endforeach()
foreach(target reo_native_stream_window_service_r4 stream_window_service_r4_inherited_tests stream_window_service_r4_quantum_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off -frounding-math)
  endif()
endforeach()
enable_testing()
add_test(NAME stream_window_service_r4_inherited COMMAND stream_window_service_r4_inherited_tests)
add_test(NAME stream_window_service_r4_quantum COMMAND stream_window_service_r4_quantum_tests)

cmake_minimum_required(VERSION 3.20)
project(reo_native_stream_window_v2 LANGUAGES CXX)
# Independent owner tests; this target never enables a game/runtime adapter.
if(NOT TARGET reo_native_adxt_shared_source)
  set(REO_NATIVE_ADXT_SHARED_SOURCE_TESTS OFF)
  add_subdirectory(../../adxt_shared_source adxt_shared_source)
endif()
if(NOT TARGET reo_native_adx_header)
  set(REO_NATIVE_ADX_HEADER_TESTS OFF)
  add_subdirectory(../../adx_header adx_header)
endif()
if(NOT TARGET reo_adx_workspace)
  add_subdirectory(../../adx_workspace adx_workspace)
endif()
if(NOT TARGET reo_adx_pcm_store_v2)
  add_subdirectory(../../adx_pcm_store/v2 adx_pcm_store_v2)
endif()
add_library(reo_native_stream_window_v2 STATIC stream_window.cpp)
target_compile_features(reo_native_stream_window_v2 PUBLIC cxx_std_20)
target_include_directories(reo_native_stream_window_v2 PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_stream_window_v2 PUBLIC reo_native_adxt_shared_source
  reo_native_adx_header reo_adx_workspace reo_adx_pcm_store_v2)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(reo_native_stream_window_v2 PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(stream_window_v2_tests stream_window_tests.cpp)
target_link_libraries(stream_window_v2_tests PRIVATE reo_native_stream_window_v2)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(stream_window_v2_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME stream_window_v2 COMMAND stream_window_v2_tests)
set_tests_properties(stream_window_v2 PROPERTIES TIMEOUT 30)

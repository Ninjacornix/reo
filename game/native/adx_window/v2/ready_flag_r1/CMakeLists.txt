cmake_minimum_required(VERSION 3.20)
project(reo_native_stream_window_v2_ready_flag_r1 LANGUAGES CXX)
# Same concrete interface/private Source2 friend; choose exactly one implementation
# in an executable. Never link the frozen and corrected owner libraries together.
foreach(dep IN ITEMS adxt_shared_source adx_header adx_workspace)
  if(NOT TARGET reo_native_${dep} AND NOT TARGET reo_${dep})
    add_subdirectory(../../../${dep} ${dep})
  endif()
endforeach()
if(NOT TARGET reo_adx_pcm_store_v2)
  add_subdirectory(../../../adx_pcm_store/v2 adx_pcm_store_v2)
endif()
add_library(reo_native_stream_window_v2_ready_flag_r1 STATIC stream_window.cpp)
target_compile_features(reo_native_stream_window_v2_ready_flag_r1 PUBLIC cxx_std_20)
target_include_directories(reo_native_stream_window_v2_ready_flag_r1 PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(reo_native_stream_window_v2_ready_flag_r1 PUBLIC reo_native_adxt_shared_source
  reo_native_adx_header reo_adx_workspace reo_adx_pcm_store_v2)
add_executable(stream_window_v2_ready_flag_r1_tests ../stream_window_tests.cpp)
target_link_libraries(stream_window_v2_ready_flag_r1_tests PRIVATE reo_native_stream_window_v2_ready_flag_r1)
foreach(target reo_native_stream_window_v2_ready_flag_r1 stream_window_v2_ready_flag_r1_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off -frounding-math)
  endif()
endforeach()
enable_testing()
add_test(NAME stream_window_v2_ready_flag_r1 COMMAND stream_window_v2_ready_flag_r1_tests)

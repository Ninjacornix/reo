cmake_minimum_required(VERSION 3.20)
project(reo_native_movie_counter_clock LANGUAGES CXX)
add_library(reo_native_movie_counter_clock STATIC clock.cpp)
target_compile_features(reo_native_movie_counter_clock PUBLIC cxx_std_20)
add_executable(movie_counter_clock_tests clock_tests.cpp)
target_link_libraries(movie_counter_clock_tests PRIVATE reo_native_movie_counter_clock)
foreach(target reo_native_movie_counter_clock movie_counter_clock_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
enable_testing()
add_test(NAME movie_counter_clock COMMAND movie_counter_clock_tests)
set_tests_properties(movie_counter_clock PROPERTIES TIMEOUT 30)

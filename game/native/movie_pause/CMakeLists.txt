cmake_minimum_required(VERSION 3.20)
project(reo_native_movie_pause LANGUAGES CXX)
add_library(reo_native_movie_pause STATIC controller.cpp)
target_compile_features(reo_native_movie_pause PUBLIC cxx_std_20)
target_include_directories(reo_native_movie_pause PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(movie_pause_tests controller_tests.cpp)
target_link_libraries(movie_pause_tests PRIVATE reo_native_movie_pause)
foreach(target reo_native_movie_pause movie_pause_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
enable_testing()
add_test(NAME movie_pause COMMAND movie_pause_tests)
set_tests_properties(movie_pause PROPERTIES TIMEOUT 30)

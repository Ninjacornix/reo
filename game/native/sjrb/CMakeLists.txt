cmake_minimum_required(VERSION 3.20)
project(reo_native_sjrb LANGUAGES CXX)
add_library(reo_native_sjrb STATIC sjrb.cpp)
target_include_directories(reo_native_sjrb PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(reo_native_sjrb PUBLIC cxx_std_20)
set_target_properties(reo_native_sjrb PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(reo_native_sjrb PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(REO_NATIVE_SJRB_TESTS "Build isolated SJRB checks" ON)
if(REO_NATIVE_SJRB_TESTS)
    enable_testing()
    add_executable(reo_native_sjrb_tests sjrb_tests.cpp)
    target_link_libraries(reo_native_sjrb_tests PRIVATE reo_native_sjrb)
    add_test(NAME sjrb_unit COMMAND reo_native_sjrb_tests)
endif()

cmake_minimum_required(VERSION 3.20)
project(reo_native_adxt_playback LANGUAGES CXX)
if(NOT TARGET reo_adx_pcm_store_v2)
  add_subdirectory(../adx_pcm_store/v2 adx_pcm_store_v2)
endif()
add_library(reo_native_adxt_playback STATIC playback.cpp)
target_compile_features(reo_native_adxt_playback PUBLIC cxx_std_20)
target_include_directories(reo_native_adxt_playback PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_adxt_playback PUBLIC reo_adx_pcm_store_v2)
add_executable(adxt_playback_tests playback_tests.cpp)
target_link_libraries(adxt_playback_tests PRIVATE reo_native_adxt_playback)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(reo_native_adxt_playback PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(adxt_playback_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME adxt_playback COMMAND adxt_playback_tests)
set_tests_properties(adxt_playback PROPERTIES TIMEOUT 30)

cmake_minimum_required(VERSION 3.20)
project(reo_native_movie_audio_clock LANGUAGES CXX)
if(NOT TARGET reo_native_movie_audio_filter)
  add_subdirectory(../movie_audio_filter movie_audio_filter)
endif()
if(NOT TARGET reo_native_movie_selected_clock)
  add_subdirectory(../movie_selected_clock movie_selected_clock)
endif()
if(NOT TARGET reo_native_adxt_control_session)
  add_subdirectory(../adxt_control_session adxt_control_session)
endif()
add_library(reo_native_movie_audio_clock STATIC clock.cpp session_source.cpp)
target_compile_features(reo_native_movie_audio_clock PUBLIC cxx_std_20)
target_include_directories(reo_native_movie_audio_clock PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_movie_audio_clock PUBLIC reo_native_movie_audio_filter reo_native_movie_selected_clock reo_native_adxt_control_session)
add_executable(movie_audio_clock_tests clock_tests.cpp)
add_executable(movie_audio_clock_owner_tests owner_tests.cpp)
target_link_libraries(movie_audio_clock_tests PRIVATE reo_native_movie_audio_clock)
target_link_libraries(movie_audio_clock_owner_tests PRIVATE reo_native_movie_audio_clock)
foreach(target reo_native_movie_audio_clock movie_audio_clock_tests movie_audio_clock_owner_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off -frounding-math)
  endif()
endforeach()
enable_testing()
add_test(NAME movie_audio_clock COMMAND movie_audio_clock_tests)
add_test(NAME movie_audio_clock_owner COMMAND movie_audio_clock_owner_tests)
set_tests_properties(movie_audio_clock movie_audio_clock_owner PROPERTIES TIMEOUT 60)

cmake_minimum_required(VERSION 3.20)
project(reo_native_adxt_attenuation LANGUAGES CXX)
add_library(reo_native_adxt_attenuation STATIC attenuation.cpp)
target_compile_features(reo_native_adxt_attenuation PUBLIC cxx_std_20)
target_include_directories(reo_native_adxt_attenuation PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(adxt_attenuation_tests attenuation_tests.cpp)
target_link_libraries(adxt_attenuation_tests PRIVATE reo_native_adxt_attenuation)
foreach(target reo_native_adxt_attenuation adxt_attenuation_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
enable_testing()
add_test(NAME adxt_attenuation COMMAND adxt_attenuation_tests)

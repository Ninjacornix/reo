cmake_minimum_required(VERSION 3.20)
project(reo_native_movie_timing_context LANGUAGES CXX)
foreach(component movie_audio_clock movie_clock_initialize movie_pause movie_admission)
  if(NOT TARGET reo_native_${component})
    add_subdirectory(../${component} ${component})
  endif()
endforeach()
add_library(reo_native_movie_timing_context STATIC context.cpp)
target_compile_features(reo_native_movie_timing_context PUBLIC cxx_std_20)
target_include_directories(reo_native_movie_timing_context PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_movie_timing_context PUBLIC reo_native_movie_audio_clock reo_native_movie_clock_initialize reo_native_movie_pause reo_native_movie_admission)
add_executable(movie_timing_context_tests context_tests.cpp)
add_executable(movie_timing_context_owner_tests owner_tests.cpp)
target_link_libraries(movie_timing_context_tests PRIVATE reo_native_movie_timing_context)
target_link_libraries(movie_timing_context_owner_tests PRIVATE reo_native_movie_timing_context)
foreach(target reo_native_movie_timing_context movie_timing_context_tests movie_timing_context_owner_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off -frounding-math)
  endif()
endforeach()
enable_testing()
add_test(NAME movie_timing_context COMMAND movie_timing_context_tests)
add_test(NAME movie_timing_context_owner COMMAND movie_timing_context_owner_tests)
set_tests_properties(movie_timing_context movie_timing_context_owner PROPERTIES TIMEOUT 60)

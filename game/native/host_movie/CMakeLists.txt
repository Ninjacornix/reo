cmake_minimum_required(VERSION 3.20)
project(reo_native_host_movie LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The first host movie implementation uses Windows Media Engine")
endif()
add_library(reo_native_host_movie STATIC player.cpp session.cpp)
target_compile_features(reo_native_host_movie PUBLIC cxx_std_20)
target_include_directories(reo_native_host_movie PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(NOT TARGET reo_native_movie_assets)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../movie_assets" "${CMAKE_CURRENT_BINARY_DIR}/movie-assets")
endif()
target_link_libraries(reo_native_host_movie PUBLIC reo_native_movie_assets mfplat mfuuid ole32 oleaut32 shlwapi d3d11 dxguid windowscodecs)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(reo_native_host_movie PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

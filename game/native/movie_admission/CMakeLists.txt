cmake_minimum_required(VERSION 3.20)
project(reo_native_movie_admission LANGUAGES CXX)
add_library(reo_native_movie_admission STATIC admission.cpp)
target_compile_features(reo_native_movie_admission PUBLIC cxx_std_20)
target_include_directories(reo_native_movie_admission PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(movie_admission_tests admission_tests.cpp)
target_link_libraries(movie_admission_tests PRIVATE reo_native_movie_admission)
foreach(target reo_native_movie_admission movie_admission_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off -frounding-math)
  endif()
endforeach()
enable_testing()
add_test(NAME movie_admission COMMAND movie_admission_tests)
set_tests_properties(movie_admission PROPERTIES TIMEOUT 30)

cmake_minimum_required(VERSION 3.20)
project(reo_native_audio_cue_queue LANGUAGES CXX)
if(NOT TARGET reo_native_adxt_instances)
    add_subdirectory(../adxt_instances adxt_instances)
endif()
if(NOT TARGET reo_native_audio_clip_loader)
    add_subdirectory(../audio_clip_loader audio_clip_loader)
endif()
add_library(reo_native_audio_cue_queue STATIC queue.cpp)
target_compile_features(reo_native_audio_cue_queue PUBLIC cxx_std_20)
target_include_directories(reo_native_audio_cue_queue PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_audio_cue_queue PUBLIC reo_native_adxt_instances reo_native_audio_clip_loader)
add_executable(audio_cue_queue_tests queue_tests.cpp)
target_link_libraries(audio_cue_queue_tests PRIVATE reo_native_audio_cue_queue)
foreach(target reo_native_audio_cue_queue audio_cue_queue_tests)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off -frounding-math)
    endif()
endforeach()
enable_testing()
add_test(NAME audio_cue_queue COMMAND audio_cue_queue_tests)
set_tests_properties(audio_cue_queue PROPERTIES TIMEOUT 60)

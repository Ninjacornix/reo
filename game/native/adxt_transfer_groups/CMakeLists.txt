cmake_minimum_required(VERSION 3.20)
project(reo_native_adxt_transfer_groups LANGUAGES CXX)
if(NOT TARGET reo_native_adxt_service_group)
  add_subdirectory(../adxt_service_group adxt_service_group)
endif()
add_library(reo_native_adxt_transfer_groups STATIC transfer_groups.cpp)
target_compile_features(reo_native_adxt_transfer_groups PUBLIC cxx_std_20)
target_include_directories(reo_native_adxt_transfer_groups PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_adxt_transfer_groups PUBLIC reo_native_adxt_service_group)
add_executable(adxt_transfer_groups_tests transfer_groups_tests.cpp)
target_link_libraries(adxt_transfer_groups_tests PRIVATE reo_native_adxt_transfer_groups)
foreach(target reo_native_adxt_transfer_groups adxt_transfer_groups_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
enable_testing()
add_test(NAME adxt_transfer_groups COMMAND adxt_transfer_groups_tests)

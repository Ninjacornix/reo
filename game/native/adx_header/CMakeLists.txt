cmake_minimum_required(VERSION 3.20)
project(reo_native_adx_header LANGUAGES CXX)
if(NOT TARGET reo_native_adxt_refill)
    set(REO_NATIVE_ADXT_REFILL_TESTS OFF)
    add_subdirectory(../adxt_refill adxt_refill)
endif()
add_library(reo_native_adx_header STATIC adx_header.cpp)
target_compile_features(reo_native_adx_header PUBLIC cxx_std_20)
target_include_directories(reo_native_adx_header PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(reo_native_adx_header PUBLIC reo_native_adxt_refill)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(reo_native_adx_header PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off -frounding-math)
endif()
option(REO_NATIVE_ADX_HEADER_TESTS "Build typed native header ownership checks" ON)
if(REO_NATIVE_ADX_HEADER_TESTS)
    enable_testing()
    add_executable(reo_native_adx_header_tests adx_header_tests.cpp)
    target_link_libraries(reo_native_adx_header_tests PRIVATE reo_native_adx_header)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(reo_native_adx_header_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off -frounding-math)
    endif()
    add_test(NAME adx_header_owner COMMAND reo_native_adx_header_tests)
endif()

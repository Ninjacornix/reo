cmake_minimum_required(VERSION 3.20)
project(reo_native_audio_mix_tail LANGUAGES CXX)
if(NOT TARGET reo_native_audio_banks_v2)
  add_subdirectory(../audio_banks/v2 audio_banks_v2)
endif()
add_library(reo_native_audio_mix_tail STATIC mixer.cpp ../../../recomp/overrides/native_banks/v3/packet_adapter.cpp)
target_compile_features(reo_native_audio_mix_tail PUBLIC cxx_std_20)
target_link_libraries(reo_native_audio_mix_tail PUBLIC reo_native_audio_banks_v2)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(reo_native_audio_mix_tail PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
endif()

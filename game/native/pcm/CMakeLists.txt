cmake_minimum_required(VERSION 3.20)
project(reo_native_pcm LANGUAGES CXX)
add_library(reo_native_pcm STATIC pcm.cpp)
target_include_directories(reo_native_pcm PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(reo_native_pcm PUBLIC cxx_std_20)
if(MSVC)
    target_compile_options(reo_native_pcm PRIVATE /fp:strict)
else()
    target_compile_options(reo_native_pcm PRIVATE -ffp-contract=off)
endif()
option(REO_NATIVE_PCM_TESTS "Build standalone PCM owner tests" ON)
if(REO_NATIVE_PCM_TESTS)
    enable_testing()
    find_package(Threads REQUIRED)
    add_executable(reo_native_pcm_tests tests/pcm_tests.cpp)
    target_link_libraries(reo_native_pcm_tests PRIVATE reo_native_pcm Threads::Threads)
    add_test(NAME pcm_owner_contract COMMAND reo_native_pcm_tests)
endif()

cmake_minimum_required(VERSION 3.20)
project(reo_native_adxt_control_owner LANGUAGES CXX)
if(NOT TARGET reo_native_adxt_instances)
  add_subdirectory(../adxt_instances adxt_instances)
endif()
add_library(reo_native_adxt_control_owner STATIC owner.cpp)
target_compile_features(reo_native_adxt_control_owner PUBLIC cxx_std_20)
target_include_directories(reo_native_adxt_control_owner PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(reo_native_adxt_control_owner PUBLIC reo_native_adxt_instances)
add_executable(adxt_control_owner_tests owner_tests.cpp)
target_link_libraries(adxt_control_owner_tests PRIVATE reo_native_adxt_control_owner)
foreach(target reo_native_adxt_control_owner adxt_control_owner_tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off -frounding-math)
  endif()
endforeach()
enable_testing()
add_test(NAME adxt_control_owner COMMAND adxt_control_owner_tests)
set_tests_properties(adxt_control_owner PROPERTIES TIMEOUT 60)

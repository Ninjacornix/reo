cmake_minimum_required(VERSION 3.20)
project(reo_native_media_import LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The first packaged movie preparation helper targets Windows")
endif()
add_executable(reo-prepare-movies prepare_movies.cpp)
target_compile_features(reo-prepare-movies PRIVATE cxx_std_20)
target_link_libraries(reo-prepare-movies PRIVATE bcrypt)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(reo-prepare-movies PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_options(reo-prepare-movies PRIVATE -municode -static)
endif()

cmake_minimum_required(VERSION 3.18)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL" CACHE STRING "Shared CRT" FORCE)
project(reo_parallel_gs_check LANGUAGES C CXX)
set(PARALLEL_GS_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../build-ninja/parallel-gs-source" CACHE PATH "Pinned paraLLEl-GS checkout")
set(PARALLEL_GS_STANDALONE ON CACHE BOOL "Library only" FORCE)
set(GRANITE_TOOLS OFF CACHE BOOL "No unrelated tools" FORCE)
set(GRANITE_INSTALL_TARGETS OFF CACHE BOOL "No install rules" FORCE)
add_subdirectory("${PARALLEL_GS_SOURCE}" parallel-gs EXCLUDE_FROM_ALL)
set(REO_PARALLEL_GS_RENDERER_SOURCE "" CACHE FILEPATH "Isolated replacement renderer implementation")
if(REO_PARALLEL_GS_RENDERER_SOURCE)
  get_target_property(reo_diag_sources parallel-gs SOURCES)
  list(REMOVE_ITEM reo_diag_sources gs_renderer.cpp)
  list(APPEND reo_diag_sources "${REO_PARALLEL_GS_RENDERER_SOURCE}")
  set_property(TARGET parallel-gs PROPERTY SOURCES "${reo_diag_sources}")
endif()
set(REO_PARALLEL_GS_INTERFACE_SOURCE "" CACHE FILEPATH "Isolated replacement GIF parser implementation")
if(REO_PARALLEL_GS_INTERFACE_SOURCE)
  get_target_property(reo_interface_sources parallel-gs SOURCES)
  list(REMOVE_ITEM reo_interface_sources gs_interface.cpp)
  list(APPEND reo_interface_sources "${REO_PARALLEL_GS_INTERFACE_SOURCE}")
  set_property(TARGET parallel-gs PROPERTY SOURCES "${reo_interface_sources}")
endif()
add_executable(parallel_gs_check main.cpp)
target_compile_features(parallel_gs_check PRIVATE cxx_std_17)
target_link_libraries(parallel_gs_check PRIVATE parallel-gs)
if(APPLE)
  enable_language(OBJC OBJCXX)
  add_executable(macos_present_check ../parallel_gs_backend/macos/present_check.mm
      ../parallel_gs_backend/macos/metal_layer.mm ../../runtime/input/macos_input.mm)
  target_compile_features(macos_present_check PRIVATE cxx_std_20)
  target_compile_options(macos_present_check PRIVATE -fobjc-arc)
  target_link_libraries(macos_present_check PRIVATE granite-vulkan
      "-framework Cocoa" "-framework Metal" "-framework QuartzCore" "-framework GameController" "-framework CoreHaptics")
endif()
if(MSVC)
  target_compile_options(parallel_gs_check PRIVATE /O2 /fp:precise /W4 /DNOMINMAX)
endif()
